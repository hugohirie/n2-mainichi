# App de estudo JLPT N2 — Plano de implementação

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** PWA pessoal para iPhone que faz o usuário praticar japonês todo dia rumo ao JLPT N2 (julho/2027): SRS de vocabulário com triagem do N3, gramática no formato da prova e leitura geradas pela API do Claude.

**Architecture:** App 100% no cliente (React + TypeScript + Vite), dados no IndexedDB (Dexie), chamadas diretas à API do Claude com a chave do usuário. A lógica fica em módulos puros e testáveis (`srs/`, `session/plan`, `ai/validate`, `lib/`) e as telas em `ui/`, que só orquestram. Os dados estáticos (vocabulário, kanji, gramática) são JSON versionados no repositório.

**Tech Stack:** React 19, TypeScript 5, Vite 8, Vitest 4, Dexie 4, ts-fsrs 5, @anthropic-ai/sdk (0.131+), zod 4, fake-indexeddb (testes), tsx (scripts), sharp (ícones).

**Spec:** [docs/superpowers/specs/2026-10-07-app-n2-design.md](../specs/2026-10-07-app-n2-design.md)

## Global Constraints

- Plataforma: PWA no iPhone, instalável via "Adicionar à Tela de Início". **Sem service worker / sem modo offline.**
- Sem servidor próprio: chamadas diretas à API (`new Anthropic({ apiKey, dangerouslyAllowBrowser: true })`).
- Modelos: `claude-sonnet-5-5` para definições/exemplos de palavras (padrão de `wordModel`), `claude-opus-5-5` para gramática e leitura (padrão de `examModel`). Opções nas Configurações: `claude-opus-5-5`, `claude-sonnet-5-5`, `claude-haiku-4-5`.
- Preços para o contador de custo (US$ por 1M tokens, entrada/saída): Opus 5.5 = 4/20, Sonnet 5.5 = 2/10, Haiku 4.5 = 1/5.
- Idioma: conteúdo de estudo em japonês (explicações em japonês simples), com tradução ao tocar. **Botões, menus e mensagens da interface em português.**
- Padrões: 10 palavras novas/dia, 1 ponto de gramática novo/dia, triagem de 30 palavras N3 por sessão.
- Furigana no texto gerado usa a notação `{漢字|かんじ}`.
- O conteúdo gerado nunca é pedido duas vezes (tudo é salvo).
- A chave da API fica só no IndexedDB e **nunca** vai no arquivo de backup.
- "Dia" = data **local** do aparelho, nunca UTC.
- Sem `React.StrictMode` (no modo dev ele roda os efeitos duas vezes e duplicaria chamadas pagas à API).
- Fontes de dados: vocabulário de `jamsinclair/open-anki-jlpt-decks` (MIT; dados do tanos.co.uk), kanji de `davidluzgouveia/kanji-data` (derivado do KANJIDIC2, CC-BY-SA 4.0). Os créditos ficam na tela de Configurações.

```ts
const pad = (n: number) => String(n).padStart(2, '0');

export function localDateKey(ms: number): string {
  const d = new Date(ms);
  return `${d.getFullYear()}-${pad(d.getMonth() + 1)}-${pad(d.getDate())}`;
}

export function monthKey(ms: number): string {
  return localDateKey(ms).slice(0, 7);
}

export function startOfLocalDay(ms: number): number {
  const d = new Date(ms);
  d.setHours(0, 0, 0, 0);
  return d.getTime();
}

export function addDays(key: string, n: number): string {
  const [y, m, d] = key.split('-').map(Number);
  return localDateKey(new Date(y, m - 1, d + n, 12).getTime());
}
```

```ts
import { createEmptyCard, fsrs, Rating, type Card, type Grade as FsrsGrade, type State } from 'ts-fsrs';

export type Grade = 'again' | 'hard' | 'good' | 'easy';

export interface SrsState {
  due: number;
  stability: number;
  difficulty: number;
  elapsedDays: number;
  scheduledDays: number;
  learningSteps: number;
  reps: number;
  lapses: number;
  state: number;
  lastReview: number | null;
}

const scheduler = fsrs({ enable_fuzz: false });

const RATING: Record<Grade, FsrsGrade> = {
  again: Rating.Again,
  hard: Rating.Hard,
  good: Rating.Good,
  easy: Rating.Easy,
};

function fromCard(c: Card): SrsState {
  return {
    due: c.due.getTime(),
    stability: c.stability,
    difficulty: c.difficulty,
    elapsedDays: c.elapsed_days,
    scheduledDays: c.scheduled_days,
    learningSteps: c.learning_steps,
    reps: c.reps,
    lapses: c.lapses,
    state: c.state,
    lastReview: c.last_review ? c.last_review.getTime() : null,
  };
}

function toCard(s: SrsState): Card {
  return {
    due: new Date(s.due),
    stability: s.stability,
    difficulty: s.difficulty,
    elapsed_days: s.elapsedDays,
    scheduled_days: s.scheduledDays,
    learning_steps: s.learningSteps,
    reps: s.reps,
    lapses: s.lapses,
    state: s.state as State,
    last_review: s.lastReview === null ? undefined : new Date(s.lastReview),
  };
}

export function newState(now: number): SrsState {
  return fromCard(createEmptyCard(new Date(now)));
}

export function review(s: SrsState, grade: Grade, now: number): SrsState {
  return fromCard(scheduler.next(toCard(s), new Date(now), RATING[grade]).card);
}

/** Palavra marcada "sei" na triagem: entra como se tivesse sido respondida "fácil". */
export function knownState(now: number): SrsState {
  return review(newState(now), 'easy', now);
}

export function isDue(s: SrsState, now: number): boolean {
  return s.due <= now;
}
```

```ts
import type { Level, VocabEntry } from '../../src/data/types';

export interface RawWord {
  expression: string;
  reading: string;
  meaning: string;
}

export function parseCsv(text: string): string[][] {
  const rows: string[][] = [];
  let row: string[] = [];
  let field = '';
  let inQuotes = false;
  const src = text.replace(/^﻿/, '');
  for (let i = 0; i < src.length; i++) {
    const ch = src[i];
    if (inQuotes) {
      if (ch === '"') {
        if (src[i + 1] === '"') {
          field += '"';
          i++;
        } else {
          inQuotes = false;
        }
      } else {
        field += ch;
      }
    } else if (ch === '"') {
      inQuotes = true;
    } else if (ch === ',') {
      row.push(field);
      field = '';
    } else if (ch === '\n') {
      row.push(field);
      rows.push(row);
      row = [];
      field = '';
    } else if (ch !== '\r') {
      field += ch;
    }
  }
  if (field !== '' || row.length > 0) {
    row.push(field);
    rows.push(row);
  }
  return rows;
}

export function rowsToWords(rows: string[][]): RawWord[] {
  const [header, ...body] = rows;
  const col = (name: string) => {
    const i = header.indexOf(name);
    if (i < 0) throw new Error(`coluna ausente no CSV: ${name}`);
    return i;
  };
  const e = col('expression');
  const r = col('reading');
  const m = col('meaning');
  return body
    .map((cells) => ({
      expression: (cells[e] ?? '').trim(),
      reading: (cells[r] ?? '').trim(),
      meaning: (cells[m] ?? '').trim(),
    }))
    .filter((w) => w.expression !== '');
}

const HAN = /\p{Script=Han}/gu;
const UNKNOWN_KANJI_FREQ = 3000;

/** Frequência do kanji mais raro da palavra (menor = mais comum). Palavras só em kana = 0. */
export function kanjiScore(expression: string, freq: Record<string, number | null>): number {
  const chars = expression.match(HAN) ?? [];
  return chars.reduce((max, ch) => Math.max(max, freq[ch] ?? UNKNOWN_KANJI_FREQ), 0);
}

export function buildVocab(n3: RawWord[], n2: RawWord[], freq: Record<string, number | null>): VocabEntry[] {
  const seen = new Set<string>();
  const out: VocabEntry[] = [];
  const levels: [Level, RawWord[]][] = [
    ['N3', n3],
    ['N2', n2],
  ];
  for (const [level, words] of levels) {
    const unique = words.filter((w) => {
      const id = `${w.expression}|${w.reading}`;
      if (seen.has(id)) return false;
      seen.add(id);
      return true;
    });
    const scored = unique.map((w, i) => ({ w, i, score: kanjiScore(w.expression, freq) }));
    scored.sort((a, b) => a.score - b.score || a.i - b.i);
    scored.forEach(({ w }, rank) =>
      out.push({ id: `${w.expression}|${w.reading}`, level, expression: w.expression, reading: w.reading, meaning: w.meaning, rank }),
    );
  }
  return out;
}

export function collectKanji(vocab: VocabEntry[]): string[] {
  const set = new Set<string>();
  for (const v of vocab) for (const ch of v.expression.match(HAN) ?? []) set.add(ch);
  return [...set].sort();
}
```

```ts
import { writeFileSync } from 'node:fs';
import { buildVocab, collectKanji, parseCsv, rowsToWords } from './lib/vocab';
import type { KanjiMap } from '../src/data/types';

const RAW = { Accept: 'application/vnd.github.raw' };
const vocabUrl = (lvl: string) => `https://api.github.com/repos/jamsinclair/open-anki-jlpt-decks/contents/src/${lvl}.csv`;
const KANJI_URL = 'https://api.github.com/repos/davidluzgouveia/kanji-data/contents/kanji.json';

interface KanjiDataEntry {
  freq: number | null;
  meanings: string[];
  readings_on: string[];
  readings_kun: string[];
}

async function get(url: string): Promise<string> {
  const res = await fetch(url, { headers: RAW });
  if (!res.ok) throw new Error(`${res.status} ao baixar ${url}`);
  return res.text();
}

async function main() {
  const [n3csv, n2csv, kanjiText] = await Promise.all([get(vocabUrl('n3')), get(vocabUrl('n2')), get(KANJI_URL)]);
  const kanjiData = JSON.parse(kanjiText) as Record<string, KanjiDataEntry>;
  const freq = Object.fromEntries(Object.entries(kanjiData).map(([k, v]) => [k, v.freq]));
  const vocab = buildVocab(rowsToWords(parseCsv(n3csv)), rowsToWords(parseCsv(n2csv)), freq);
  const kanji: KanjiMap = {};
  for (const ch of collectKanji(vocab)) {
    const k = kanjiData[ch];
    if (k) kanji[ch] = { meanings: k.meanings, on: k.readings_on, kun: k.readings_kun };
  }
  writeFileSync('src/data/vocab.json', JSON.stringify(vocab));
  writeFileSync('src/data/kanji.json', JSON.stringify(kanji));
  const n3 = vocab.filter((v) => v.level === 'N3').length;
  console.log(`vocab: ${vocab.length} (N3 ${n3}, N2 ${vocab.length - n3}); kanji: ${Object.keys(kanji).length}`);
}

main().catch((e) => {
  console.error(e);
  process.exit(1);
});
```

```ts
import type { N2DB } from './db';

const TABLES = ['items', 'wordDetails', 'exercises', 'days', 'settings', 'usage'] as const;
type TableName = (typeof TABLES)[number];

export interface Backup {
  app: 'n2-study';
  version: 1;
  exportedAt: number;
  tables: Record<TableName, unknown[]>;
}

export class BackupError extends Error {}

export async function exportBackup(db: N2DB, now: number): Promise<Backup> {
  const tables = {} as Record<TableName, unknown[]>;
  for (const name of TABLES) tables[name] = await db.table(name).toArray();
  tables.settings = (tables.settings as { key: string }[]).filter((r) => r.key !== 'apiKey');
  return { app: 'n2-study', version: 1, exportedAt: now, tables };
}

export function parseBackup(text: string): Backup {
  let data: unknown;
  try {
    data = JSON.parse(text);
  } catch {
    throw new BackupError('O arquivo não é um JSON válido.');
  }
  const b = data as Partial<Backup> | null;
  if (!b || b.app !== 'n2-study') throw new BackupError('Este arquivo não é um backup deste app.');
  if (b.version !== 1) throw new BackupError(`Versão de backup não suportada: ${String(b.version)}`);
  for (const name of TABLES) {
    if (!Array.isArray(b.tables?.[name])) throw new BackupError(`Backup incompleto: falta a tabela ${name}.`);
  }
  return b as Backup;
}

export async function importBackup(db: N2DB, backup: Backup): Promise<void> {
  await db.transaction('rw', TABLES.map((n) => db.table(n)), async () => {
    const apiKey = await db.settings.get('apiKey');
    for (const name of TABLES) {
      await db.table(name).clear();
      await db.table(name).bulkPut(backup.tables[name]);
    }
    if (apiKey) await db.settings.put(apiKey);
  });
}
```

```ts
import type { GrammarSet, Reading, ReadingLength } from './schemas';

interface Mcq {
  options: string[];
  answerIndex: number;
  explanationsJa: string[];
}

function checkMcq(q: Mcq, label: string): string[] {
  const errs: string[] = [];
  if (q.options.length !== 4) errs.push(`${label}: precisa de 4 alternativas`);
  if (q.options.some((o) => o.trim() === '')) errs.push(`${label}: alternativa vazia`);
  if (new Set(q.options.map((o) => o.trim())).size !== q.options.length) errs.push(`${label}: alternativas repetidas`);
  if (!Number.isInteger(q.answerIndex) || q.answerIndex < 0 || q.answerIndex >= q.options.length) {
    errs.push(`${label}: answerIndex fora do intervalo`);
  }
  if (q.explanationsJa.length !== q.options.length) errs.push(`${label}: precisa de uma explicação por alternativa`);
  return errs;
}

export function validateGrammarSet(s: GrammarSet): string[] {
  const errs: string[] = [];
  if (s.questions.length < 2 || s.questions.length > 3) errs.push('precisa de 2 a 3 questões');
  if (!s.explanationJa.trim()) errs.push('explicação do ponto vazia');
  s.questions.forEach((q, i) => {
    const label = `questão ${i + 1}`;
    errs.push(...checkMcq(q, label));
    if (!q.stem.trim()) errs.push(`${label}: enunciado vazio`);
    if (q.type === 'order' && (q.stem.match(/★/g) ?? []).length !== 1) errs.push(`${label}: ordenação precisa de exatamente um ★`);
    if (q.type === 'form' && !/（\s*）|\(\s*\)/.test(q.stem)) errs.push(`${label}: lacuna （　） ausente`);
  });
  return errs;
}

const LENGTH_RULES: Record<ReadingLength, { min: number; max: number; questions: number }> = {
  short: { min: 120, max: 350, questions: 1 },
  medium: { min: 350, max: 800, questions: 3 },
};

export function validateReading(r: Reading, length: ReadingLength): string[] {
  const rule = LENGTH_RULES[length];
  const errs: string[] = [];
  const size = r.passage.trim().length;
  if (size < rule.min || size > rule.max) errs.push(`tamanho do texto fora do esperado (${size} caracteres)`);
  if (r.questions.length !== rule.questions) errs.push(`precisa de ${rule.questions} perguntas`);
  r.questions.forEach((q, i) => {
    errs.push(...checkMcq(q, `pergunta ${i + 1}`));
    if (!q.question.trim()) errs.push(`pergunta ${i + 1}: enunciado vazio`);
  });
  return errs;
}
```

```ts
export type RubySegment = { text: string; ruby?: string };

const RUBY = /\{([^|{}]+)\|([^{}]+)\}/g;

export function parseRuby(input: string): RubySegment[] {
  const out: RubySegment[] = [];
  let last = 0;
  for (const m of input.matchAll(RUBY)) {
    if (m.index > last) out.push({ text: input.slice(last, m.index) });
    out.push({ text: m[1], ruby: m[2] });
    last = m.index + m[0].length;
  }
  if (last < input.length) out.push({ text: input.slice(last) });
  return out;
}
```

```ts
export type GlossSegment = { text: string; entry?: number };

export function splitByGlossary(passage: string, words: string[]): GlossSegment[] {
  const ordered = words
    .map((w, i) => ({ w, i }))
    .filter((x) => x.w.length > 0)
    .sort((a, b) => b.w.length - a.w.length);
  const out: GlossSegment[] = [];
  let buf = '';
  let pos = 0;
  while (pos < passage.length) {
    const hit = ordered.find(({ w }) => passage.startsWith(w, pos));
    if (hit) {
      if (buf) out.push({ text: buf });
      buf = '';
      out.push({ text: hit.w, entry: hit.i });
      pos += hit.w.length;
    } else {
      buf += passage[pos];
      pos++;
    }
  }
  if (buf) out.push({ text: buf });
  return out;
}
```

```ts
import { zodOutputFormat } from '@anthropic-ai/sdk/helpers/zod';
import type { z } from 'zod';

export type Effort = 'low' | 'medium' | 'high';

export interface LlmCall<T> {
  model: string;
  system: string;
  user: string;
  schema: z.ZodType<T>;
  effort: Effort;
  maxTokens: number;
}

export interface LlmResult<T> {
  data: T | null;
  inputTokens: number;
  outputTokens: number;
}

export type Llm = <T>(call: LlmCall<T>) => Promise<LlmResult<T>>;

export function createLlm(apiKey: string): Llm {
  // A chave é do próprio usuário e fica só no aparelho dele (ver spec, seção 7).
  const client = new Anthropic({ apiKey, dangerouslyAllowBrowser: true });
  return async <T>(call: LlmCall<T>): Promise<LlmResult<T>> => {
    const res = await client.messages.parse({
      model: call.model,
      max_tokens: call.maxTokens,
      system: call.system,
      messages: [{ role: 'user', content: call.user }],
      output_config: { effort: call.effort, format: zodOutputFormat(call.schema) },
    });
    const refused = res.stop_reason === 'refusal';
    return {
      data: refused ? null : ((res.parsed_output as T | null) ?? null),
      inputTokens: res.usage.input_tokens,
      outputTokens: res.usage.output_tokens,
    };
  };
}
```

```ts
import Anthropic from '@anthropic-ai/sdk';

export type AiErrorKind = 'auth' | 'billing' | 'network' | 'rate' | 'server' | 'unknown';

export class AiError extends Error {
  constructor(
    readonly kind: AiErrorKind,
    message: string,
  ) {
    super(message);
    this.name = 'AiError';
  }
}

export function classifyError(e: unknown): AiErrorKind {
  if (e instanceof AiError) return e.kind;
  if (e instanceof Anthropic.APIConnectionError || e instanceof TypeError) return 'network';
  const { status, message } = (e ?? {}) as { status?: unknown; message?: unknown };
  if (typeof status === 'number') {
    if (status === 401 || status === 403) return 'auth';
    if (status === 429) return 'rate';
    if (status >= 500) return 'server';
    if (status === 400 && typeof message === 'string' && /credit balance/i.test(message)) return 'billing';
  }
  return 'unknown';
}

  network: 'Sem conexão com a API. Tente de novo em instantes.',
  rate: 'Limite de uso da API atingido. Tente de novo mais tarde.',
  server: 'A API está com problemas no momento. Tente de novo mais tarde.',
  unknown: 'Erro inesperado ao falar com a API.',
};
```

```ts
import type { GrammarPoint, VocabEntry } from '../data/types';
import type { Llm, LlmCall, LlmResult } from './client';
import { AiError, classifyError } from './errors';
import { grammarPrompt, readingPrompt, wordDetailsPrompt } from './prompts';
import {
  GrammarSetSchema,
  ReadingSchema,
  WordDetailsBatchSchema,
  type GrammarSet,
  type Reading,
  type ReadingLength,
} from './schemas';
import { validateGrammarSet, validateReading } from './validate';

export interface AiDeps {
  llm: Llm;
  wordModel: string;
  examModel: string;
  onUsage(model: string, inputTokens: number, outputTokens: number): Promise<void>;
}

export interface WordDetails {
  definitionJa: string;
  example: string;
}

const MAX_ATTEMPTS = 2;

async function callValidated<T>(deps: AiDeps, call: LlmCall<T>, validate: (data: T) => string[]): Promise<T | null> {
  for (let attempt = 0; attempt < MAX_ATTEMPTS; attempt++) {
    let res: LlmResult<T>;
    try {
      res = await deps.llm(call);
    } catch (e) {
      throw new AiError(classifyError(e), e instanceof Error ? e.message : String(e));
    }
    await deps.onUsage(call.model, res.inputTokens, res.outputTokens);
    if (res.data !== null && validate(res.data).length === 0) return res.data;
  }
  return null;
}

export async function generateWordDetails(deps: AiDeps, words: VocabEntry[]): Promise<Map<string, WordDetails>> {
  const wanted = new Set(words.map((w) => w.id));
  const { system, user } = wordDetailsPrompt(words);
  const data = await callValidated(
    deps,
    { model: deps.wordModel, system, user, schema: WordDetailsBatchSchema, effort: 'low', maxTokens: 8000 },
    (d) => (d.words.some((w) => wanted.has(w.id)) ? [] : ['nenhuma palavra pedida veio na resposta']),
  );
  const out = new Map<string, WordDetails>();
  for (const w of data?.words ?? []) {
    const definitionJa = w.definitionJa.trim();
    const example = w.example.trim();
    if (wanted.has(w.id) && definitionJa && example) out.set(w.id, { definitionJa, example });
  }
  return out;
}

export async function generateGrammarSet(deps: AiDeps, point: GrammarPoint): Promise<GrammarSet | null> {
  const { system, user } = grammarPrompt(point);
  return callValidated(
    deps,
    { model: deps.examModel, system, user, schema: GrammarSetSchema, effort: 'medium', maxTokens: 16000 },
    validateGrammarSet,
  );
}

export async function generateReading(deps: AiDeps, length: ReadingLength, targetWords: string[]): Promise<Reading | null> {
  const { system, user } = readingPrompt(length, targetWords);
  return callValidated(
    deps,
    { model: deps.examModel, system, user, schema: ReadingSchema, effort: 'medium', maxTokens: 16000 },
    (r) => validateReading(r, length),
  );
}
```

```ts
import type { N2DB } from '../storage/db';
import { monthKey } from '../lib/dates';

/** US$ por 1M tokens. */
export const PRICES: Record<string, { input: number; output: number }> = {
  'claude-opus-5-5': { input: 4, output: 20 },
  'claude-sonnet-5-5': { input: 2, output: 10 },
  'claude-haiku-4-5': { input: 1, output: 5 },
};

export function costUsd(model: string, inputTokens: number, outputTokens: number): number {
  const p = PRICES[model];
  if (!p) return 0;
  return (inputTokens * p.input + outputTokens * p.output) / 1_000_000;
}

export async function recordUsage(db: N2DB, now: number, model: string, inputTokens: number, outputTokens: number): Promise<void> {
  const month = monthKey(now);
  const id = `${month}|${model}`;
  await db.transaction('rw', db.usage, async () => {
    const row = await db.usage.get(id);
    await db.usage.put({
      id,
      month,
      model,
      inputTokens: (row?.inputTokens ?? 0) + inputTokens,
      outputTokens: (row?.outputTokens ?? 0) + outputTokens,
    });
  });
}

export async function monthUsage(db: N2DB, now: number) {
  const month = monthKey(now);
  const rows = await db.usage.filter((r) => r.month === month).toArray();
  return rows.map((r) => ({
    model: r.model,
    inputTokens: r.inputTokens,
    outputTokens: r.outputTokens,
    costUsd: costUsd(r.model, r.inputTokens, r.outputTokens),
  }));
}
```

```ts
import { addDays } from '../lib/dates';

/** Dias seguidos com a sessão mínima cumprida. Se hoje ainda não foi feito, conta a partir de ontem. */
export function computeStreak(doneDays: Iterable<string>, today: string): number {
  const done = new Set(doneDays);
  let day = done.has(today) ? today : addDays(today, -1);
  let n = 0;
  while (done.has(day)) {
    n++;
    day = addDays(day, -1);
  }
  return n;
}
```

```ts
export const BACKLOG_PAUSE_NEW = 80;
export const MIN_GRAMMAR = 3;
export const CONTINUE_NEW_WORDS = 5;
export const CONTINUE_TRIAGE = 20;

export interface PlanInput {
  dueWordItemIds: string[];
  dueGrammarIds: string[];
  untriagedN3Ids: string[];
  unintroducedN2Ids: string[];
  unintroducedGrammarIds: string[];
  introducedToday: { newWords: number; triage: number; grammar: number };
  settings: { newWordsPerDay: number; newGrammarPerDay: number; triagePerSession: number };
}

export interface DailyPlan {
  reviews: string[];
  triage: string[];
  newWords: string[];
  grammar: string[];
  reading: boolean;
}

const remaining = (quota: number, used: number) => Math.max(0, quota - used);

export function buildDailyPlan(p: PlanInput): DailyPlan {
  const backlog = p.dueWordItemIds.length > BACKLOG_PAUSE_NEW;
  const triaging = p.untriagedN3Ids.length > 0;
  const newWordQuota = backlog || triaging ? 0 : remaining(p.settings.newWordsPerDay, p.introducedToday.newWords);
  const newGrammar = p.unintroducedGrammarIds.slice(0, remaining(p.settings.newGrammarPerDay, p.introducedToday.grammar));
  return {
    reviews: p.dueWordItemIds.slice(0, MIN_REVIEW_CAP),
    triage: p.untriagedN3Ids.slice(0, remaining(p.settings.triagePerSession, p.introducedToday.triage)),
    newWords: p.unintroducedN2Ids.slice(0, newWordQuota),
    grammar: [...p.dueGrammarIds, ...newGrammar].slice(0, MIN_GRAMMAR),
    reading: false,
  };
}

export function buildContinuePlan(p: PlanInput): DailyPlan {
  const triaging = p.untriagedN3Ids.length > 0;
  return {
    reviews: p.dueWordItemIds.slice(0, MIN_REVIEW_CAP),
    triage: triaging ? p.untriagedN3Ids.slice(0, CONTINUE_TRIAGE) : [],
    newWords: triaging ? [] : p.unintroducedN2Ids.slice(0, CONTINUE_NEW_WORDS),
    grammar: p.dueGrammarIds.slice(0, MIN_GRAMMAR),
    reading: true,
  };
}
```

```ts
import type { DailyPlan } from './plan';

export type Step =
  | { kind: 'review'; itemId: string }
  | { kind: 'triage'; vocabId: string }
  | { kind: 'new'; vocabId: string }
  | { kind: 'grammar'; grammarId: string }
  | { kind: 'reading' };

export function planToSteps(plan: DailyPlan): Step[] {
  return [
    ...plan.reviews.map((itemId): Step => ({ kind: 'review', itemId })),
    ...plan.triage.map((vocabId): Step => ({ kind: 'triage', vocabId })),
    ...plan.newWords.map((vocabId): Step => ({ kind: 'new', vocabId })),
    ...plan.grammar.map((grammarId): Step => ({ kind: 'grammar', grammarId })),
    ...(plan.reading ? [{ kind: 'reading' } as Step] : []),
  ];
}

export function planVocabIds(plan: DailyPlan): string[] {
  const reviewWords = plan.reviews.filter((id) => id.startsWith('word:')).map((id) => id.slice('word:'.length));
  return [...reviewWords, ...plan.triage, ...plan.newWords];
}
```

```ts
export async function loadPlanInput(ctx: Ctx): Promise<PlanInput> {
  const now = ctx.now();
  const dayStart = startOfLocalDay(now);
  const settings = await getSettings(ctx.db);
  const items = await ctx.db.items.toArray();
  const ids = new Set(items.map((i) => i.id));
  const active = items.filter((i) => !i.discarded);
  const due = active.filter((i) => i.due <= now).sort((a, b) => a.due - b.due);
  const today = items.filter((i) => i.introducedAt >= dayStart);
  return {
    dueWordItemIds: due.filter((i) => i.kind === 'word').map((i) => i.id),
    dueGrammarIds: due.filter((i) => i.kind === 'grammar').map((i) => i.refId),
    untriagedN3Ids: vocab.filter((v) => v.level === 'N3' && !ids.has(itemId('word', v.id))).map((v) => v.id),
    unintroducedN2Ids: vocab.filter((v) => v.level === 'N2' && !ids.has(itemId('word', v.id))).map((v) => v.id),
    unintroducedGrammarIds: grammar.filter((g) => !ids.has(itemId('grammar', g.id))).map((g) => g.id),
    introducedToday: {
      newWords: today.filter((i) => i.kind === 'word' && i.origin === 'new').length,
      triage: today.filter((i) => i.kind === 'word' && i.origin !== 'new').length,
      grammar: today.filter((i) => i.kind === 'grammar').length,
    },
    settings: {
      newWordsPerDay: settings.newWordsPerDay,
      newGrammarPerDay: settings.newGrammarPerDay,
      triagePerSession: settings.triagePerSession,
    },
  };

export async function introduceWord(ctx: Ctx, vocabId: string, origin: ItemOrigin): Promise<ItemRow> {
  const entry = vocabById.get(vocabId);
  if (!entry) throw new Error(`palavra desconhecida: ${vocabId}`);
  const id = itemId('word', vocabId);
  const existing = await ctx.db.items.get(id);
  if (existing) return existing;
  const now = ctx.now();
  const srs = origin === 'triage-known' ? knownState(now) : newState(now);
  const row: ItemRow = { id, kind: 'word', refId: vocabId, level: entry.level, origin, srs, due: srs.due, introducedAt: now, discarded: false };
  await ctx.db.items.put(row);
  return row;
}

export function answerTriage(ctx: Ctx, vocabId: string, known: boolean): Promise<ItemRow> {
  return introduceWord(ctx, vocabId, known ? 'triage-known' : 'triage-unknown');
}

export async function ensureWordDetails(ctx: Ctx, vocabIds: string[]): Promise<AiErrorKind | null> {
  if (!ctx.ai) return null;
  const unique = [...new Set(vocabIds)];
  const existing = await ctx.db.wordDetails.bulkGet(unique);
  const missing = unique
    .filter((_, i) => !existing[i])
    .map((id) => vocabById.get(id))
    .filter((v): v is VocabEntry => v !== undefined);
  for (let i = 0; i < missing.length; i += WORD_BATCH) {
    try {
      const map = await generateWordDetails(ctx.ai, missing.slice(i, i + WORD_BATCH));
      const createdAt = ctx.now();
      await ctx.db.wordDetails.bulkPut([...map].map(([wordId, d]) => ({ wordId, ...d, createdAt })));
    } catch (e) {
      if (e instanceof AiError) return e.kind;
      throw e;
    }
  }
  return null;
}

export async function getWordView(ctx: Ctx, vocabId: string): Promise<WordView> {
  const entry = vocabById.get(vocabId);
  if (!entry) throw new Error(`palavra desconhecida: ${vocabId}`);
  return { entry, details: await ctx.db.wordDetails.get(vocabId) };
}

export async function answerReview(ctx: Ctx, id: string, grade: Grade): Promise<ItemRow> {
  const item = await ctx.db.items.get(id);
  if (!item) throw new Error(`item inexistente: ${id}`);
  const srs = review(item.srs, grade, ctx.now());
  const updated = { ...item, srs, due: srs.due };
  await ctx.db.items.put(updated);
  return updated;
}

async function runGeneration<T>(ctx: Ctx, generate: (ai: AiDeps) => Promise<T | null>): Promise<{ ok: true; value: T } | { ok: false; error: StartError }> {
  if (!ctx.ai) return { ok: false, error: 'no-ai' };
  try {
    const value = await generate(ctx.ai);
    return value === null ? { ok: false, error: 'invalid' } : { ok: true, value };
  } catch (e) {
    if (e instanceof AiError) return { ok: false, error: e.kind };
    throw e;
  }
}

export async function startGrammar(ctx: Ctx, grammarId: string): Promise<StartResult<GrammarSet>> {
  const point = grammarById.get(grammarId);
  if (!point) throw new Error(`gramática desconhecida: ${grammarId}`);
  const gen = await runGeneration(ctx, (ai) => generateGrammarSet(ai, point));
  if (!gen.ok) return gen;
  const now = ctx.now();
  const id = itemId('grammar', grammarId);
  if (!(await ctx.db.items.get(id))) {
    const srs = newState(now);
    await ctx.db.items.put({ id, kind: 'grammar', refId: grammarId, level: point.level, origin: 'new', srs, due: srs.due, introducedAt: now, discarded: false });
  }
  const exerciseId = await ctx.db.exercises.add({
    kind: 'grammar',
    refId: grammarId,
    content: gen.value,
    answers: null,
    correctCount: null,
    srsBefore: null,
    createdAt: now,
    answeredAt: null,
    discarded: false,
  });
  return { ok: true, exerciseId, content: gen.value };
}

function countCorrect(questions: { answerIndex: number }[], answers: number[]): number {
  return questions.filter((q, i) => answers[i] === q.answerIndex).length;
}

export async function answerGrammar(ctx: Ctx, exerciseId: number, answers: number[]) {
  const ex = await ctx.db.exercises.get(exerciseId);
  if (!ex || ex.kind !== 'grammar' || !ex.refId) throw new Error(`exercício de gramática inexistente: ${exerciseId}`);
  const set = ex.content as GrammarSet;
  const correct = countCorrect(set.questions, answers);
  const total = set.questions.length;
  const grade = gradeFromResults(correct, total);
  const now = ctx.now();
  await ctx.db.transaction('rw', ctx.db.items, ctx.db.exercises, async () => {
    const item = await ctx.db.items.get(itemId('grammar', ex.refId!));
    if (!item) throw new Error(`item de gramática inexistente: ${ex.refId}`);
    const srs = review(item.srs, grade, now);
    await ctx.db.exercises.update(exerciseId, { answers, correctCount: correct, srsBefore: item.srs, answeredAt: now });
    await ctx.db.items.put({ ...item, srs, due: srs.due });
  });
  return { correct, total, grade };
}

export async function startReading(ctx: Ctx): Promise<StartResult<Reading>> {
  const done = await ctx.db.exercises.where('kind').equals('reading').count();
  const length: ReadingLength = done % 2 === 0 ? 'short' : 'medium';
  const recent = (await ctx.db.items.where('kind').equals('word').toArray())
    .filter((i) => i.origin !== 'triage-known' && !i.discarded)
    .sort((a, b) => b.introducedAt - a.introducedAt)
    .slice(0, READING_TARGET_WORDS)
    .map((i) => vocabById.get(i.refId)?.expression)
    .filter((w): w is string => w !== undefined);
  const gen = await runGeneration(ctx, (ai) => generateReading(ai, length, recent));
  if (!gen.ok) return gen;
  const exerciseId = await ctx.db.exercises.add({
    kind: 'reading',
    refId: null,
    content: gen.value,
    answers: null,
    correctCount: null,
    srsBefore: null,
    createdAt: ctx.now(),
    answeredAt: null,
    discarded: false,
  });
  return { ok: true, exerciseId, content: gen.value };
}

export async function answerReading(ctx: Ctx, exerciseId: number, answers: number[]) {
  const ex = await ctx.db.exercises.get(exerciseId);
  if (!ex || ex.kind !== 'reading') throw new Error(`leitura inexistente: ${exerciseId}`);
  const reading = ex.content as Reading;
  const correct = countCorrect(reading.questions, answers);
  await ctx.db.exercises.update(exerciseId, { answers, correctCount: correct, answeredAt: ctx.now() });
  return { correct, total: reading.questions.length };
}

export async function discardExercise(ctx: Ctx, exerciseId: number): Promise<void> {
  await ctx.db.transaction('rw', ctx.db.items, ctx.db.exercises, async () => {
    const ex = await ctx.db.exercises.get(exerciseId);
    if (!ex) return;
    await ctx.db.exercises.update(exerciseId, { discarded: true });
    if (ex.kind === 'grammar' && ex.refId && ex.srsBefore) {
      const item = await ctx.db.items.get(itemId('grammar', ex.refId));
      if (item) await ctx.db.items.put({ ...item, srs: ex.srsBefore, due: ex.srsBefore.due });
    }
  });
}

export async function markMinimumDone(ctx: Ctx): Promise<void> {
  await ctx.db.days.put({ date: localDateKey(ctx.now()), minimumDone: true });

export async function isMinimumDoneToday(ctx: Ctx): Promise<boolean> {

}

  );
}

export async function getProgress(ctx: Ctx): Promise<Progress> {

  const words = items.filter((i) => i.kind === 'word');
```

```ts
/** Abre a folha de compartilhamento do iOS (salvar em Arquivos) ou baixa o arquivo. Retorna false se o usuário cancelou. */
export async function shareOrDownload(name: string, text: string): Promise<boolean> {
  const file = new File([text], name, { type: 'application/json' });
  if (navigator.canShare?.({ files: [file] })) {
    try {
      await navigator.share({ files: [file] });
      return true;
    } catch (e) {
      if ((e as Error).name === 'AbortError') return false;
    }
  }
  const url = URL.createObjectURL(file);
  const a = document.createElement('a');
  a.href = url;
  a.download = name;
  a.click();
  setTimeout(() => URL.revokeObjectURL(url), 1000);
  return true;
}
```

```tsx
// src/ui/App.tsx
import { useCallback, useEffect, useMemo, useState } from 'react';
import { db } from '../storage/db';
import { getSettings, type Settings } from '../storage/settings';
import { createLlm } from '../ai/client';
import { recordUsage } from '../ai/usage';
import type { Ctx } from '../session/service';
import { SetupScreen } from './SetupScreen';
import { TodayScreen } from './TodayScreen';
import { SessionScreen } from './SessionScreen';
import { ProgressScreen } from './ProgressScreen';
import { SettingsScreen } from './SettingsScreen';

type Screen = { name: 'today' } | { name: 'session'; mode: 'minimum' | 'continue'; key: number } | { name: 'progress' } | { name: 'settings' };

export default function App() {
  const [settings, setSettings] = useState<Settings | null>(null);
  const [screen, setScreen] = useState<Screen>({ name: 'today' });

  const reload = useCallback(async () => setSettings(await getSettings(db)), []);
  useEffect(() => {
    void reload();
  }, [reload]);

  const ctx = useMemo<Ctx | null>(() => {
    if (!settings) return null;
    const ai = settings.apiKey
      ? {
          llm: createLlm(settings.apiKey),
          wordModel: settings.wordModel,
          examModel: settings.examModel,
          onUsage: (model: string, inputTokens: number, outputTokens: number) => recordUsage(db, Date.now(), model, inputTokens, outputTokens),
        }
      : null;
    return { db, ai, now: () => Date.now() };
  }, [settings]);

  if (!settings || !ctx) return <div className="screen center">読み込み中…</div>;
  if (!settings.apiKey) return <SetupScreen onSaved={reload} />;

  const goToday = () => setScreen({ name: 'today' });
  const startSession = (mode: 'minimum' | 'continue') => setScreen({ name: 'session', mode, key: Date.now() });

  switch (screen.name) {
    case 'session':
      return <SessionScreen key={screen.key} ctx={ctx} mode={screen.mode} onExit={goToday} onContinue={() => startSession('continue')} />;
    case 'progress':
      return <ProgressScreen ctx={ctx} onBack={goToday} />;
    case 'settings':
      return <SettingsScreen ctx={ctx} settings={settings} onSaved={reload} onBack={goToday} />;
    default:
      return <TodayScreen ctx={ctx} settings={settings} onStart={startSession} onNav={(name) => setScreen({ name })} />;
  }
}
```

```tsx
import { useEffect, useState } from 'react';
import { AI_ERROR_MESSAGES } from '../ai/errors';
import { planToSteps, planVocabIds, type Step } from '../session/steps';
import { ensureWordDetails, getContinuePlan, getTodayPlan, type Ctx } from '../session/service';
import { WordCard } from './WordCard';
import { TriageCard } from './TriageCard';
import { GrammarExercise } from './GrammarExercise';
import { ReadingExercise } from './ReadingExercise';
import { SessionDone } from './SessionDone';

export interface SessionProps {
  ctx: Ctx;
  mode: 'minimum' | 'continue';
  onExit: () => void;
  onContinue: () => void;
}

export function SessionScreen({ ctx, mode, onExit, onContinue }: SessionProps) {
  const [steps, setSteps] = useState<Step[] | null>(null);
  const [index, setIndex] = useState(0);
  const [warning, setWarning] = useState<string | null>(null);

  useEffect(() => {
    void (async () => {
      const plan = mode === 'minimum' ? await getTodayPlan(ctx) : await getContinuePlan(ctx);
      const err = await ensureWordDetails(ctx, planVocabIds(plan));
      if (err) setWarning(AI_ERROR_MESSAGES[err]);
      setSteps(planToSteps(plan));
    })();
  }, [ctx, mode]);

  if (!steps) return <div className="screen center">Preparando a sessão…</div>;
  if (index >= steps.length) return <SessionDone ctx={ctx} mode={mode} onExit={onExit} onContinue={onContinue} />;

  const next = (extra: Step[] = []) => {
    if (extra.length > 0) setSteps((s) => [...s!, ...extra]);
    setIndex((i) => i + 1);
  };
  const step = steps[index];

  return (
    <div className="screen">
      <div className="topbar">
        <button className="link" onClick={onExit}>Sair</button>
        <span>{index + 1} / {steps.length}</span>
      </div>
      {warning && <p className="bad">{warning}</p>}
      {step.kind === 'review' && <WordCard key={index} ctx={ctx} vocabId={step.itemId.slice('word:'.length)} introduce={false} onDone={() => next()} />}
      {step.kind === 'new' && <WordCard key={index} ctx={ctx} vocabId={step.vocabId} introduce onDone={() => next()} />}
      {step.kind === 'triage' && (
        <TriageCard key={index} ctx={ctx} vocabId={step.vocabId} onDone={(known) => next(known ? [] : [{ kind: 'review', itemId: `word:${step.vocabId}` }])} />
      )}
      {step.kind === 'grammar' && <GrammarExercise key={index} ctx={ctx} grammarId={step.grammarId} onDone={() => next()} />}
      {step.kind === 'reading' && <ReadingExercise key={index} ctx={ctx} onDone={() => next()} />}
    </div>
  );
}
```

As palavras marcadas "não sei" na triagem entram no fim da fila como revisão. As definições delas já foram buscadas antes, porque `planVocabIds` inclui as palavras da triagem.

```tsx
import { useCallback, useEffect, useState } from 'react';
import { grammarById } from '../data';
import type { GrammarSet } from '../ai/schemas';
import { answerGrammar, discardExercise, startGrammar, type Ctx } from '../session/service';
import { errorText } from './errorText';

interface Props {
  ctx: Ctx;
  grammarId: string;
  onDone: () => void;
}

type Phase =
  | { name: 'loading' }
  | { name: 'error'; message: string }
  | { name: 'answering'; exerciseId: number; set: GrammarSet; answers: number[] }
  | { name: 'result'; exerciseId: number; set: GrammarSet; answers: number[]; correct: number };

export function GrammarExercise({ ctx, grammarId, onDone }: Props) {
  const point = grammarById.get(grammarId)!;
  const [phase, setPhase] = useState<Phase>({ name: 'loading' });
  const [showPt, setShowPt] = useState(false);
  const [busy, setBusy] = useState(false);

  const load = useCallback(async () => {
    setPhase({ name: 'loading' });
    const r = await startGrammar(ctx, grammarId);
    setPhase(r.ok ? { name: 'answering', exerciseId: r.exerciseId, set: r.content, answers: [] } : { name: 'error', message: errorText(r.error) });
  }, [ctx, grammarId]);

  useEffect(() => {
    void load();
  }, [load]);

  const choose = async (option: number) => {
    if (phase.name !== 'answering' || busy) return;
    const answers = [...phase.answers, option];
    if (answers.length < phase.set.questions.length) {
      setPhase({ ...phase, answers });
      return;
    }
    setBusy(true);
    const { correct } = await answerGrammar(ctx, phase.exerciseId, answers);
    setBusy(false);
    setPhase({ name: 'result', exerciseId: phase.exerciseId, set: phase.set, answers, correct });
  };

  const report = async (exerciseId: number) => {
    await discardExercise(ctx, exerciseId);
    onDone();
  };

  const header = (
    <p className="muted">
      文法 {point.level}: <b>{point.pattern}</b>
    </p>
  );

  if (phase.name === 'loading') return <div className="card">{header}<p>Gerando exercício…</p></div>;
  if (phase.name === 'error')
    return (
      <div className="card">
        {header}
        <p className="bad">{phase.message}</p>
        <div className="row">
          <button onClick={() => void load()}>Tentar de novo</button>
          <button onClick={onDone}>Pular</button>
        </div>
      </div>
    );

  if (phase.name === 'answering') {
    const i = phase.answers.length;
    const q = phase.set.questions[i];
    return (
      <div className="card">
        {header}
        <p className="muted">Questão {i + 1}/{phase.set.questions.length} — {q.type === 'order' ? 'qual peça vai no ★?' : 'escolha a melhor opção'}</p>
        <p className="passage">{q.stem}</p>
        {q.options.map((o, k) => (
          <button key={k} className="option" disabled={busy} onClick={() => void choose(k)}>
            {k + 1}. {o}
          </button>
        ))}
        <button className="link" onClick={() => void report(phase.exerciseId)}>Questão com problema</button>
      </div>
    );
  }

  return (
    <div className="card">
      {header}
      <p className={phase.correct === phase.set.questions.length ? 'ok' : 'bad'}>
        Acertou {phase.correct} de {phase.set.questions.length}
      </p>
      <p>{phase.set.explanationJa}</p>
      {showPt ? <p className="muted">{phase.set.explanationPt}</p> : <button className="link" onClick={() => setShowPt(true)}>Ver em português</button>}
      {phase.set.questions.map((q, i) => (
        <div key={i} className="popover">
          <p className="passage">{q.stem}</p>
          {q.options.map((o, k) => (
            <div key={k}>
              <p className={k === q.answerIndex ? 'ok' : k === phase.answers[i] ? 'bad' : 'muted'}>
                {k === q.answerIndex ? '✓' : k === phase.answers[i] ? '✗' : '・'} {o}
              </p>
              <p className="muted">{q.explanationsJa[k]}</p>
            </div>
          ))}
          {showPt && <p className="muted">PT: {q.translationPt}</p>}
        </div>
      ))}
      <div className="row">
        <button onClick={() => void report(phase.exerciseId)}>Questão com problema</button>
        <button className="primary" onClick={onDone}>Próximo</button>
      </div>
    </div>
  );
}
```

```tsx
import { useCallback, useEffect, useState } from 'react';
import type { Reading } from '../ai/schemas';
import { splitByGlossary } from '../lib/glossary';
import { answerReading, discardExercise, startReading, type Ctx } from '../session/service';
import { errorText } from './errorText';

interface Props {
  ctx: Ctx;
  onDone: () => void;
}

type Phase =
  | { name: 'loading' }
  | { name: 'error'; message: string }
  | { name: 'answering'; exerciseId: number; reading: Reading; answers: number[] }
  | { name: 'result'; exerciseId: number; reading: Reading; answers: number[]; correct: number };

export function ReadingExercise({ ctx, onDone }: Props) {
  const [phase, setPhase] = useState<Phase>({ name: 'loading' });
  const [gloss, setGloss] = useState<number | null>(null);
  const [busy, setBusy] = useState(false);

  const load = useCallback(async () => {
    setPhase({ name: 'loading' });
    const r = await startReading(ctx);
    setPhase(r.ok ? { name: 'answering', exerciseId: r.exerciseId, reading: r.content, answers: [] } : { name: 'error', message: errorText(r.error) });
  }, [ctx]);

  useEffect(() => {
    void load();
  }, [load]);

  if (phase.name === 'loading') return <div className="card"><p>Gerando texto de leitura…</p></div>;
  if (phase.name === 'error')
    return (
      <div className="card">
        <p className="bad">{phase.message}</p>
        <div className="row">
          <button onClick={() => void load()}>Tentar de novo</button>
          <button onClick={onDone}>Pular</button>
        </div>
      </div>
    );

  const { reading, exerciseId } = phase;
  const words = reading.glossary.map((g) => g.word);
  const entry = gloss !== null ? reading.glossary[gloss] : null;

  const choose = async (option: number) => {
    if (phase.name !== 'answering' || busy) return;
    const answers = [...phase.answers, option];
    if (answers.length < reading.questions.length) {
      setPhase({ ...phase, answers });
      return;
    }
    setBusy(true);
    const { correct } = await answerReading(ctx, exerciseId, answers);
    setBusy(false);
    setPhase({ name: 'result', exerciseId, reading, answers, correct });
  };

  const report = async () => {
    await discardExercise(ctx, exerciseId);
    onDone();
  };

  return (
    <div className="card">
      <p className="muted">読解</p>
      <h2>{reading.title}</h2>
      <p className="passage">
        {splitByGlossary(reading.passage, words).map((s, i) =>
          s.entry !== undefined ? (
            <span key={i} className="gloss" onClick={() => setGloss(s.entry === gloss ? null : s.entry!)}>
              {s.text}
            </span>
          ) : (
            <span key={i}>{s.text}</span>
          ),
        )}
      </p>
      {entry && (
        <div className="popover">
          <p>
            <b>{entry.word}</b>（{entry.reading}）
          </p>
          <p className="muted">{entry.meaningJa}</p>
        </div>
      )}

      {phase.name === 'answering' ? (
        (() => {
          const i = phase.answers.length;
          const q = reading.questions[i];
          return (
            <>
              <p className="muted">Pergunta {i + 1}/{reading.questions.length}</p>
              <p>{q.question}</p>
              {q.options.map((o, k) => (
                <button key={k} className="option" disabled={busy} onClick={() => void choose(k)}>
                  {k + 1}. {o}
                </button>
              ))}
            </>
          );
        })()
      ) : (
        <>
          <p className={phase.correct === reading.questions.length ? 'ok' : 'bad'}>
            Acertou {phase.correct} de {reading.questions.length}
          </p>
          {reading.questions.map((q, i) => (
            <div key={i} className="popover">
              <p>{q.question}</p>
              {q.options.map((o, k) => (
                <div key={k}>
                  <p className={k === q.answerIndex ? 'ok' : k === phase.answers[i] ? 'bad' : 'muted'}>
                    {k === q.answerIndex ? '✓' : k === phase.answers[i] ? '✗' : '・'} {o}
                  </p>
                  <p className="muted">{q.explanationsJa[k]}</p>
                </div>
              ))}
            </div>
          ))}
        </>
      )}
      <div className="row">
        <button onClick={() => void report()}>Questão com problema</button>
        {phase.name === 'result' && <button className="primary" onClick={onDone}>Próximo</button>}
      </div>
    </div>
  );
}
```
